Google Apps Script Web App Code (.gs)

Salin kode Google Apps Script di bawah ini ke editor **Google Apps Script** (script.google.com) milik Google Sheets Anda, lalu Publish sebagai **Web App** dengan akses "Anyone". Tempelkan URL Web App ke variabel JavaScript `APPS_SCRIPT_URL` di website. 

// Google Apps Script Backend for ARFA Studio
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Append Form Entry row to Google Sheet
    sheet.appendRow([
      new Date(),
      data.name,
      data.phone,
      data.category,
      data.location,
      data.message
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      "result": "success",
      "message": "Data berhasil tersimpan ke Google Sheets ARFA Studio"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      "result": "error",
      "error": error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
